```csharp
using System.Text.RegularExpressions;

namespace VideoTriage.Core.Verify;

public static class FfmpegStderrFilter
{
    // Matches the -f null muxer's VFR timestamp gripe — benign, not corruption.
    private static readonly Regex BenignDts = new(
        @"non.?monotonically increasing dts",
        RegexOptions.Compiled | RegexOptions.IgnoreCase);

    // Matches ffmpeg's repeat-collapse suffix — benign.
    private static readonly Regex BenignRepeat = new(
        @"Last message repeated \d+ times",
        RegexOptions.Compiled | RegexOptions.IgnoreCase);

    /// <summary>
    /// Returns the stderr lines that represent real decode errors.
    /// Blank lines and lines matching the two benign patterns are excluded.
    /// Any other non-empty line is treated as a real error.
    /// </summary>
    public static IReadOnlyList<string> RealErrorLines(string stderrText)
    {
        if (string.IsNullOrWhiteSpace(stderrText))
            return [];

        var lines = stderrText.Split(new[] { "\r\n", "\n" }, StringSplitOptions.None);
        var errors = new List<string>();

        foreach (var line in lines)
        {
            var trimmed = line.Trim();
            if (trimmed.Length == 0)
                continue;
            if (BenignDts.IsMatch(trimmed))
                continue;
            if (BenignRepeat.IsMatch(trimmed))
                continue;
            errors.Add(trimmed);
        }

        return errors;
    }
}
```

```csharp
namespace VideoTriage.Core.Verify;

public static class ResolutionParity
{
    /// <summary>
    /// Returns true when the output resolution matches the source within <paramref name="tolerancePercent"/>.
    /// Comparison is orientation-agnostic: each pair is sorted into (min, max) before comparison,
    /// so a W/H swap (e.g. a rotated phone video) passes.
    /// </summary>
    public static bool Matches(int srcW, int srcH, int outW, int outH, double tolerancePercent)
    {
        // Sort both pairs into (min, maj)
        var (srcMin, srcMaj) = srcW <= srcH ? (srcW, srcH) : (srcH, srcW);
        var (outMin, outMaj) = outW <= outH ? (outW, outH) : (outH, outW);

        return WithinTol(srcMin, outMin, tolerancePercent)
            && WithinTol(srcMaj, outMaj, tolerancePercent);
    }

    private static bool WithinTol(int src, int output, double tolerancePercent)
    {
        if (src == 0)
            return output == 0;

        var diffFraction = Math.Abs(output - src) / (double)src;
        return diffFraction * 100.0 <= tolerancePercent;
    }
}
```

```csharp
namespace VideoTriage.Core.Verify;

public static class DurationParity
{
    /// <summary>
    /// Returns true when |output - source| / source &lt;= tolerancePercent / 100.
    /// If source is zero, only a zero output passes (divide-by-zero guard).
    /// </summary>
    public static bool WithinTolerance(TimeSpan source, TimeSpan output, double tolerancePercent)
    {
        var srcSeconds = source.TotalSeconds;

        if (srcSeconds == 0)
            return output.TotalSeconds == 0;

        var diffFraction = Math.Abs(output.TotalSeconds - srcSeconds) / srcSeconds;
        return diffFraction * 100.0 <= tolerancePercent;
    }
}
```

The verifier runs checks in this exact order (first failure short-circuits):
1. File exists and `size > 0` → else `MissingOrEmpty`
2. `ffprobe` the output; any probe failure, including no video stream, yields `ProbeFailed`
3. `DurationParity.WithinTolerance` → else `DurationMismatch`
4. If `options.RequireResolutionMatch` and `!ResolutionParity.Matches` → `ResolutionMismatch`

```csharp
using VideoTriage.Core.Models;
using VideoTriage.Core.Probing;
using VideoTriage.Core.Tools;

namespace VideoTriage.Core.Verify;

/// <summary>
/// Orchestrates verification of an encoded output file:
///   1. File exists and is non-empty
///   2. ffprobe succeeds and finds a video stream
///   3. Duration parity within DurationTolerancePercent
///   4. Resolution parity (if RequireResolutionMatch) within ResolutionTolerancePercent
///   5. Audio parity (if RequireAudioParity and source has audio)
///   6. Deep decode via ffmpeg -f null (if DeepVerify) — stderr filtered for real errors only
/// </summary>
public sealed class OutputVerifier(
    string ffmpegPath,
    IProcessRunner runner,
    IFfprobeService ffprobe) : IOutputVerifier
{
    public async Task<VerificationResult> VerifyAsync(
        VideoStats source,
        string outputPath,
        TriageOptions options,
        CancellationToken ct = default)
    {
        // 1. File must exist and be non-empty
        if (!File.Exists(outputPath) || new FileInfo(outputPath).Length == 0)
        {
            return Fail(VerificationOutcome.MissingOrEmpty,
                $"Output file is missing or empty: {outputPath}");
        }

        // 2. Probe the output with ffprobe
        var probeResult = await ffprobe.ProbeAsync(outputPath, ct);
        if (!probeResult.Succeeded || probeResult.Stats is null)
        {
            var msg = probeResult.Failure?.Message ?? "ffprobe returned no video stats";
            return Fail(VerificationOutcome.ProbeFailed, $"Output probe failed: {msg}");
        }

        var outputStats = probeResult.Stats;

        // 3. Duration parity
        if (!DurationParity.WithinTolerance(source.Duration, outputStats.Duration, options.DurationTolerancePercent))
        {
            return Fail(VerificationOutcome.DurationMismatch,
                $"Duration mismatch: source={source.Duration.TotalSeconds:0.##}s " +
                $"output={outputStats.Duration.TotalSeconds:0.##}s " +
                $"tolerance={options.DurationTolerancePercent}%");
        }

        // 4. Resolution parity (optional)
        if (options.RequireResolutionMatch &&
            !ResolutionParity.Matches(
                source.Width, source.Height,
                outputStats.Width, outputStats.Height,
                options.ResolutionTolerancePercent))
        {
            return Fail(VerificationOutcome.ResolutionMismatch,
                $"Resolution mismatch: source={source.Width}x{source.Height} " +
                $"output={outputStats.Width}x{outputStats.Height} " +
                $"tolerance={options.ResolutionTolerancePercent}%");
        }

        // 5. Audio parity (optional — only when source has audio)
        if (options.RequireAudioParity && source.HasAudio && !outputStats.HasAudio)
        {
            return Fail(VerificationOutcome.AudioMissing,
                "Output is missing audio track that was present in the source.");
        }

        // 6. Deep decode via ffmpeg (optional)
        if (options.DeepVerify)
        {
            var decodeError = await RunDeepDecodeAsync(outputPath, ct);
            if (decodeError is not null)
            {
                return Fail(VerificationOutcome.DecodeError,
                    $"Deep decode found errors: {decodeError}");
            }
        }

        return new VerificationResult
        {
            Outcome = VerificationOutcome.Valid,
            Reason = "All verification checks passed.",
            OutputStats = outputStats
        };
    }

    /// <summary>
    /// Runs  ffmpeg -nostdin -v error -i &lt;outputPath&gt; -f null -
    /// and returns the first real error line, or null if the decode is clean.
    /// Stderr is captured to a temp file via ProcessRunner (never via shell redirect).
    /// </summary>
    private async Task<string?> RunDeepDecodeAsync(string outputPath, CancellationToken ct)
    {
        var request = new ProcessRequest
        {
            FileName = ffmpegPath,
            Arguments = ["-nostdin", "-v", "error", "-i", outputPath, "-f", "null", "-"],
            StderrDirectory = Path.GetTempPath(),
            Timeout = TimeSpan.FromMinutes(30)  // long videos can take time to decode
        };

        var processResult = await runner.RunAsync(request, ct);

        // Read the stderr file that ProcessRunner wrote
        var stderrText = File.Exists(processResult.StandardErrorPath)
            ? File.ReadAllText(processResult.StandardErrorPath)
            : string.Empty;

        var realErrors = FfmpegStderrFilter.RealErrorLines(stderrText);
        if (realErrors.Count > 0)
        {
            return realErrors[0];
        }

        return processResult.Succeeded ? null : $"ffmpeg exited {processResult.ExitCode}";
    }

    private static VerificationResult Fail(VerificationOutcome outcome, string reason) =>
        new() { Outcome = outcome, Reason = reason };
}
```
